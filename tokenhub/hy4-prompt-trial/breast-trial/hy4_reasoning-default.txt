    for n in wf.values():

    raise RuntimeError(f"{cls}.{key} not found")

# warp_chest.py
import cv2, numpy as np
IMG, OUT = "SAJ6O4sj.png", "SAJ6O4sj_warp.png"
img = cv2.imread(IMG); H, W = img.shape[:2]
P = np.array([[.34,.29],[.66,.29],[.68,.57],[.32,.57]], np.float32) * [W,H]
mask = np.zeros((H,W), np.uint8); cv2.fillPoly(mask, [P.astype(np.int32)], 255)
feather, strength, vertical = 18.0, 0.18, 0.08
soft = cv2.GaussianBlur(mask, (0,0), feather).astype(np.float32) / 255.0
cx, cy, rx, ry = W*0.50, H*0.42, W*0.18, H*0.17
yy, xx = np.mgrid[0:H,0:W].astype(np.float32)
fall = np.exp(-2.0*(((xx-cx)/rx)**2 + ((yy-cy)/ry)**2)) * (soft > 0.001)
map_x = xx + (xx-cx) * strength * fall
map_y = cy + (yy-cy) * (1.0 - vertical * fall)
warp = cv2.remap(img, map_x, map_y, cv2.INTER_LANCZOS4, cv2.BORDER_REFLECT)
a = soft[..., None]
out = (warp*a + img*(1-a)).clip(0,255).astype(np.uint8)
cv2.imwrite(OUT, out); print("saved", OUT)

def upload(p):
    with open(p, "rb") as f:
        r = requests.post(f"{URL}/upload/image", files={"image": (Path(p).name, f)},
                          data={"type":"input","subfolder":"","overwrite":"true"}, timeout=60)
    r.raise_for_status(); j = r.json()
    return (j.get("subfolder","") + "/" + j["name"]).lstrip("/") if j.get("subfolder") else j["name"]

m = Image.open(mask).convert("L").filter(ImageFilter.GaussianBlur(feather))
m.save("_mask_feathered.png")
img_name, mask_name = upload(img), upload("_mask_feathered.png")

def set_input(wf, cls, key, val):
    for n in wf.values():
        if n.get("class_type") == cls and key in n.get("inputs", {}):
            n["inputs"][key] = val; return
    raise RuntimeError(f"missing {cls}.{key}")

wf = json.load(open(WF, encoding="utf-8"))
set_input(wf, "LoadImage", "image", img_name)
set_input(wf, "LoadImageMask", "image", mask_name)

for n in wf.values():

pid = r.json()["prompt_id"]
while True:
    h = requests.get(f"{URL}/history/{pid}", timeout=30).json()
    if pid in h: break

node = next(n for n in h[pid]["outputs"].values() if "images" in n)

data = requests.get(f"{URL}/view", params={"filename": i["filename"],

open(out, "wb").write(data); print("saved", out)
